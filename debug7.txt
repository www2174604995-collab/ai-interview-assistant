function loadResume(){
  if(!token)return;
  fetch(API+"/api/resume",{headers:{"Authorization":"Bearer "+token}}).then(function(r){return r.json()}).then(function(d){
    if(d.code===200&&d.data&&d.data.hasResume){
      document.getElementById("resumeStatus").style.display="";
      document.getElementById("resumeStatus").innerHTML="已上传简历，共 "+d.data.chars+" 个字符";
      document.getElementById("resumeStatus").style.borderLeft="3px solid var(--success)";
      document.getElementById("resumePreview").style.display="";
      document.getElementById("resumePreviewText").textContent=d.data.preview+"...";
      document.getElementById("resumeDeleteBtn").style.display="";
    }
  });
}

  var idx={auth:0,interview:1,history:2,report:3,resume:4,settings:5}[name];
  if (idx !== undefined) { document.querySelectorAll(".tab")[idx].classList.add("active"); document.getElementById("tab-"+name).classList.add("active"); }
  if(name==="history")loadHistory();
  if(name==="resume")setTimeout(loadResume,100);
  if(name==="settings")setTimeout(loadApiKeyStatus,100);
}
