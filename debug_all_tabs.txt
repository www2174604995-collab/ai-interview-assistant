var API = "";
var token = localStorage.getItem("jwt_token") || "";
var currentSessionId = null;
var historySessionId = null;
updateUI();

function updateUI() {
  var b = document.getElementById("loginBadge"), d = document.getElementById("tokenDisplay"), o = document.getElementById("logoutBtn");
  if (token) { b.textContent = "已登录"; b.className = "badge badge-on"; d.value = token.substring(0,15)+"..."; o.style.display = ""; }
  else { b.textContent = "未登录"; b.className = "badge badge-off"; d.value = "-"; o.style.display = "none"; }
}

function toast(msg, t) {
  var el = document.createElement("div"); el.className = "toast-item toast-"+t; el.textContent = msg;
  document.getElementById("toast").appendChild(el);
  setTimeout(function(){ el.style.transition="opacity 0.3s"; el.style.opacity="0"; setTimeout(function(){el.remove()},300); }, 3000);
}

function api(m, p, b) {
  var o = {method:m, headers:{"Content-Type":"application/json"}};
  if (token) o.headers["Authorization"] = "Bearer "+token;
  if (b) o.body = JSON.stringify(b);
  return fetch(API+p, o).then(function(r){return r.json()})["catch"](function(){return{code:500,message:"网络错误"}});
}

function setToken(t){token=t;localStorage.setItem("jwt_token",t);updateUI();}
function logout(){token="";localStorage.removeItem("jwt_token");updateUI();toast("已退出登录","info");}
function register(){api("POST","/api/auth/register",{username:document.getElementById("regUsername").value,email:document.getElementById("regEmail").value,password:document.getElementById("regPassword").value}).then(function(d){if(d.code===200){setToken(d.data.token);toast("注册成功! "+d.data.username,"success")}else{toast("注册失败: "+(d.message||"未知错误"),"error")}})}
function login(){api("POST","/api/auth/login",{username:document.getElementById("loginUsername").value,password:document.getElementById("loginPassword").value}).then(function(d){if(d.code===200){setToken(d.data.token);toast("登录成功! "+d.data.username,"success")}else{toast("登录失败: "+(d.message||"用户名或密码错误"),"error")}})}

function startInterview(){
  if(!token){toast("请先登录","error");return}
  // 检查是否有简历，有则提示正在分析
  fetch(API+"/api/resume",{headers:{"Authorization":"Bearer "+token}}).then(function(r){return r.json()}).then(function(d){if(d.code===200&&d.data&&d.data.hasResume){toast("正在分析简历并生成面试问题，请稍候...","info")}});
  toast("正在开始面试...","info");
  api("POST","/api/interview/start",{position:document.getElementById("startPosition").value,difficulty:document.getElementById("startDifficulty").value}).then(function(d){
    if(d.code===200){currentSessionId=d.data.sessionId;document.getElementById("chatSessionId").value=currentSessionId;document.getElementById("chatHeader").textContent="对话面板 - 会话 #"+currentSessionId;var box=document.getElementById("chatBox");box.innerHTML="";addMsg("ai",d.data.reply);toast("面试已开始! 会话 #"+currentSessionId,"success")}
    else{toast("失败: "+(d.message||"未知错误"),"error")}})
}

function sendChat(){
  var sid=document.getElementById("chatSessionId").value,msg=document.getElementById("chatMessage").value;
  if(!sid){toast("请先开始面试","error");return}
  if(!msg.trim()){toast("请输入回答","error");return}
  addMsg("user",msg);document.getElementById("chatMessage").value="";
  var btn=document.getElementById("sendBtn");btn.textContent="思考中...";btn.disabled=true;
  api("POST","/api/interview/chat",{sessionId:parseInt(sid),message:msg}).then(function(d){
    btn.textContent="发送";btn.disabled=false;document.getElementById("chatMessage").focus();
    if(d.code===200&&d.data){addMsg("ai",d.data.reply);if(d.data.completed)setTimeout(function(){toast("面试已结束! 可生成报告","info")},500)}
    else{addMsg("ai","错误: "+(d.message||"发送失败"))}})
}


function loadResume(){
  if(!token)return;
  fetch(API+"/api/resume",{headers:{"Authorization":"Bearer "+token}}).then(function(r){return r.json()}).then(function(d){
    if(d.code===200&&d.data&&d.data.hasResume){
      document.getElementById("resumeStatus").style.display="";
      document.getElementById("resumeStatus").innerHTML="已上传简历，共 "+d.data.chars+" 个字符";
      document.getElementById("resumeStatus").style.borderLeft="3px solid var(--success)";
      document.getElementById("resumePreview").style.display="";
      document.getElementById("resumePreviewText").textContent=d.data.preview+"...";
      document.getElementById("resumeDeleteBtn").style.display="";
    }
  });
}
