function uploadFile(file){
  if(!file)return;
  var ext=file.name.toLowerCase();
  if(!ext.endsWith(".pdf")&&!ext.endsWith(".docx")){toast("\u4ec5\u652f\u6301 PDF \u548c DOCX \u683c\u5f0f","error");return}
  if(file.size>5*1024*1024){toast("\u6587\u4ef6\u8d85\u8fc7 5MB \u9650\u5236","error");return}
  if(!token){toast("\u8bf7\u5148\u767b\u5f55","error");return}
  var st=document.getElementById("resumeStatus");st.style.display="";st.innerHTML="\u6b63\u5728\u4e0a\u4f20\u5e76\u89e3\u6790...";st.style.borderLeft="3px solid var(--primary)";
  var fd=new FormData();fd.append("file",file);
  fetch(API+"/api/resume/upload",{method:"POST",headers:{"Authorization":"Bearer "+token},body:fd}).then(function(r){return r.json()}).then(function(d){
    if(d.code===200){st.innerHTML="\u4e0a\u4f20\u6210\u529f! \u5df2\u63d0\u53d6 "+d.data.chars+" \u4e2a\u5b57\u7b26";st.style.borderLeft="3px solid var(--success)";document.getElementById("resumePreview").style.display="";document.getElementById("resumePreviewText").textContent=d.data.preview+"...";document.getElementById("resumeDeleteBtn").style.display="";toast("\u7b80\u5386\u5df2\u66f4\u65b0!","success")}
