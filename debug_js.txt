function switchTab(name){
  document.querySelectorAll(".tab").forEach(function(t){t.classList.remove("active")});
  document.querySelectorAll(".tab-content").forEach(function(t){t.classList.remove("active")});
  var idx={auth:0,interview:1,history:2,report:3,resume:4,settings:5}[name];
  if (idx !== undefined) { document.querySelectorAll(".tab")[idx].classList.add("active"); document.getElementById("tab-"+name).classList.add("active"); }
  if(name==="history")loadHistory();
  if(name==="resume")setTimeout(loadResume,100);
  if(name==="settings")setTimeout(loadApiKeyStatus,100);
}

function loadHistory(){
  if(!token){toast("请先登录","error");return}
  api("GET","/api/interview/sessions").then(function(d){
    var el=document.getElementById("historyList");
    if(!d.data||d.data.length===0){el.innerHTML='<div style="color:var(--muted);padding:30px;text-align:center;">暂无记录</div>';return}
    el.innerHTML=d.data.map(function(s){
      var st=s.status==="COMPLETED"?"[完成]":"[进行]";
      var t=new Date(s.createdAt).toLocaleString("zh-CN");
      return '<div class="history-item" onclick="viewSession('+s.id+')"><div class="left"><div class="title">'+(s.position||"未知")+'</div><div class="meta">'+s.difficulty+" - "+(s.totalTurns||0)+"轮 - "+(s.messageCount||0)+"条</div></div><div class=\"right\">"+st+"<br>"+t+"</div></div>"}).join("")})
}

function loadResume(){

      document.getElementById("resumePreview").style.display="";
      document.getElementById("resumePreviewText").textContent=d.data.preview+"...";
      document.getElementById("resumeDeleteBtn").style.display="";
    }
  });
}

function loadApiKeyStatus() {
  if (!token) return;
  fetch(API + "/api/user/api-key", {
    headers: { "Authorization": "Bearer " + token }
  }).then(function(r) { return r.json(); }).then(function(d) {
    if (d.code === 200 && d.data && d.data.hasKey) {
      var st = document.getElementById("apiKeyStatus");

function saveApiKey() {
  var key = document.getElementById("apiKeyInput").value.trim();
  if (!key) { toast("\u8bf7\u8f93\u5165API Key", "error"); return; }
  if (key.indexOf("sk-") !== 0) { toast("API Key \u5e94\u4ee5 sk- \u5f00\u5934", "error"); return; }
  if (!token) { toast("\u8bf7\u5148\u767b\u5f55", "error"); return; }

  var btn = document.querySelector('#tab-settings .btn-primary');
  btn.textContent = "\u4fdd\u5b58\u4e2d..."; btn.disabled = true;

  api("POST", "/api/user/api-key", { apiKey: key }).then(function(d) {
    btn.textContent = "\u4fdd\u5b58"; btn.disabled = false;
    if (d.code === 200) {
      var status = document.getElementById("apiKeyStatus");
      status.style.display = "";
      status.innerHTML = "\u5df2\u4fdd\u5b58 " + d.data.masked;
      status.style.borderLeft = "3px solid var(--success)";
      document.getElementById("apiKeyInput").value = "";
      toast("API Key \u5df2\u4fdd\u5b58", "success");
    } else {
      toast("\u4fdd\u5b58\u5931\u8d25: " + (d.message || "\u672a\u77e5\u9519\u8bef"), "error");
    }
  });
}
